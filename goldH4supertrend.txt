//@version=6
strategy("Gold H4 SuperTrend v2 (trail)", overlay=true, initial_capital=10000, default_qty_type=strategy.fixed, default_qty_value=1, commission_type=strategy.commission.cash_per_contract, commission_value=0.2, slippage=10, pyramiding=0)

// ---- Inputs ----
atrLen   = input.int(10, "SuperTrend ATR length", minval=1, group="SuperTrend")
factor   = input.float(3.0, "SuperTrend factor", minval=0.5, step=0.25, group="SuperTrend")
riskPct  = input.float(2.0, "Risk per trade, % of equity", minval=0.1, step=0.25, group="Sizing")
maxLev   = input.float(3.0, "Max leverage (position value / equity)", minval=0.5, step=0.5, group="Sizing")
useMgmt  = input.bool(true, "Use trade management", group="Management")
beR      = input.float(1.0, "Move stop to breakeven after (R)", minval=0.25, step=0.25, group="Management")
tpR      = input.float(0.0, "Take profit at (R), 0 = off", minval=0.0, step=0.5, group="Management")
trailOn  = input.bool(true, "ATR trailing stop after breakeven", group="Management")
trailATR = input.float(3.0, "Trailing distance (x ATR)", minval=0.5, step=0.25, group="Management")

// ---- Signals ----
[st, dir] = ta.supertrend(factor, atrLen)
atr  = ta.atr(atrLen)
buy  = ta.change(dir) < 0
sell = ta.change(dir) > 0

riskDist = math.max(math.abs(close - st), syminfo.mintick * 10)
qty = math.min(strategy.equity * riskPct / 100 / riskDist, strategy.equity * maxLev / close)

// ---- Orders ----
var float entryPx = na
var float riskD   = na
var bool beHit    = false
var float hh      = na
var float ll      = na

if buy
    strategy.entry("Long", strategy.long, qty=qty)
    entryPx := close
    riskD := riskDist
    beHit := false
    hh := high
    ll := low
if sell
    strategy.entry("Short", strategy.short, qty=qty)
    entryPx := close
    riskD := riskDist
    beHit := false
    hh := high
    ll := low

if useMgmt
    if strategy.position_size > 0 and not sell
        hh := math.max(hh, high)
        if high >= entryPx + beR * riskD
            beHit := true
        float stopL = st
        if beHit
            stopL := math.max(st, entryPx)
            if trailOn
                stopL := math.max(stopL, hh - trailATR * atr)
        float tpL = tpR > 0 ? entryPx + tpR * riskD : na
        strategy.exit("Long Exit", "Long", stop=stopL, limit=tpL)
    if strategy.position_size < 0 and not buy
        ll := math.min(ll, low)
        if low <= entryPx - beR * riskD
            beHit := true
        float stopS = st
        if beHit
            stopS := math.min(st, entryPx)
            if trailOn
                stopS := math.min(stopS, ll + trailATR * atr)
        float tpS = tpR > 0 ? entryPx - tpR * riskD : na
        strategy.exit("Short Exit", "Short", stop=stopS, limit=tpS)

// ---- Plots ----
plot(st, "SuperTrend", color = dir < 0 ? color.green : color.red, linewidth=2)
plotshape(buy, "Buy", shape.triangleup, location.belowbar, color.green, size=size.small)
plotshape(sell, "Sell", shape.triangledown, location.abovebar, color.red, size=size.small)
