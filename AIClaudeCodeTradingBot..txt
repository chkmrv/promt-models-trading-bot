blk1Start = input.time(timestamp("1 Jan 2000 00:00 +0000"), "Blackout 1 start (e.g. FOMC/CPI)", group = G6)

// ═══════════════════════════ PERSISTENT STATE ════════════════════════════════
var float peakEq       = strategy.initial_capital
var bool  halted       = false
var float dayStartEq   = strategy.initial_capital
var int   entriesToday = 0
var int   consecLoss   = 0
var int   cooldownUntil = -1
var float npAtEntry    = 0.0
var bool  tradeLive    = false
var int   signalBar    = -1
var int   curDir       = 0
var float stopDist     = na
var float initQty      = 0.0
var int   stage        = 0
var float curStop      = na
var float extremeClose = na
var float lastStopSent = na
var float tp1Line      = na
var float tp2Line      = na

var table dash = table.new(position.top_right, 2, 11, bgcolor = color.new(color.black, 15), border_width = 1, border_color = color.new(color.gray, 60))

// ═══════════════════════════════ HELPERS ═════════════════════════════════════
f_lot(float x) =>
    math.round(math.floor(x / lotStep + 1e-9) * lotStep, 8)

f_msg(string action, float qty, float px, float sl, float t1, float t2) =>
    string s = '{"passphrase":"' + passphrase + '","action":"' + action + '","symbol":"' + syminfo.ticker + '"'
    s += ',"qty":' + str.tostring(nz(qty), "0.00000")
    s += ',"price":' + str.tostring(nz(px), "0.00")
    s += ',"stop":' + str.tostring(nz(sl), "0.00")
    s += ',"tp1":' + str.tostring(nz(t1), "0.00")
    s += ',"tp2":' + str.tostring(nz(t2), "0.00")
    s += ',"order_id":"ITE-' + str.tostring(time) + '-' + action + '","timestamp":' + str.tostring(time) + '}'
    s

f_row(int r, string k, string v, color c) =>
    table.cell(dash, 0, r, k, text_color = color.silver, text_halign = text.align_left, text_size = size.small)
    table.cell(dash, 1, r, v, text_color = c, text_halign = text.align_left, text_size = size.small)

// ═════════════════════ INDICATORS (all at global scope) ══════════════════════
float atr14   = ta.atr(atrLen)
float atrT    = ta.atr(trailAtrLen)
float atrPct  = atr14 / close * 100
float ema21   = ta.ema(close, emaPullLen)
[diPlus, diMinus, adxVal] = ta.dmi(adxLen, adxLen)
float rsiVal  = ta.rsi(close, rsiLen)
float volMA   = ta.sma(volume, volLen)
float donHi   = ta.highest(high, donEntryLen)[1]
float donLo   = ta.lowest(low, donEntryLen)[1]
float donFastHi = ta.highest(high, donFastLen)[1]
float donFastLo = ta.lowest(low, donFastLen)[1]
float donExitLo = ta.lowest(low, donExitLen)[1]
float donExitHi = ta.highest(high, donExitLen)[1]
float lowestN  = ta.lowest(low, pullLook)
float highestN = ta.highest(high, pullLook)
int barsSinceRsiUp   = ta.barssince(ta.crossover(rsiVal, 50))
int barsSinceRsiDown = ta.barssince(ta.crossunder(rsiVal, 50))

// Regime (4H): only the last CLOSED higher-timeframe bar is used ([1] + lookahead_on = no repaint)
[dClose, dEmaF, dEmaS, dEmaSPrev] = request.security(syminfo.tickerid, regimeTF,
     [close[1], ta.ema(close, emaFastLen)[1], ta.ema(close, emaSlowLen)[1], ta.ema(close, emaSlowLen)[1 + slopeLen]],
     lookahead = barmerge.lookahead_on)
// Macro (1D)
[mClose, mEma] = request.security(syminfo.tickerid, macroTF,
     [close[1], ta.ema(close, macroLen)[1]],
     lookahead = barmerge.lookahead_on)

bool macroData = not useMacro or (not na(mClose) and not na(mEma))
bool dataOK = bar_index >= warmupBars and macroData and not na(dClose) and not na(dEmaF) and not na(dEmaS) and not na(dEmaSPrev) and not na(atr14) and atr14 > 0 and not na(adxVal) and not na(rsiVal) and not na(donHi) and not na(volMA)
bool macroLong   = not useMacro or mClose > mEma
bool macroShort  = not useMacro or mClose < mEma
bool regimeLong  = dataOK and dClose > dEmaS and dEmaF > dEmaS and dEmaS > dEmaSPrev and macroLong
bool regimeShort = dataOK and dClose < dEmaS and dEmaF < dEmaS and dEmaS < dEmaSPrev and macroShort

// Stop distance and cost-vs-risk (recomputed every bar; also shown on dashboard)
float distNow      = math.min(math.max(stopAtrMult * atr14, close * minStopPct / 100), close * maxStopPct / 100)
float distPctNow   = distNow / close * 100
float costPctNow   = 2 * feePct + 2 * slipTicks * syminfo.mintick / close * 100
float costShareNow = distPctNow > 0 ? costPctNow / distPctNow * 100 : 999.0

// ═════════════════════════ RISK STATE / BREAKERS ═════════════════════════════
int  dayIdx = math.floor(time / 86400000)
bool newDay = dayIdx != nz(dayIdx[1], dayIdx)
if newDay
    dayStartEq   := strategy.equity
    entriesToday := 0

if time >= resumeTime and nz(time[1], 0) < resumeTime
    peakEq := strategy.equity
    halted := false

peakEq := math.max(peakEq, strategy.equity)
float ddPct      = peakEq > 0 ? (peakEq - strategy.equity) / peakEq * 100 : 0.0
float dayLossNow = dayStartEq > 0 ? (dayStartEq - strategy.equity) / dayStartEq * 100 : 0.0
bool  dayLocked  = dayLossNow >= dailyLossPct

if ddPct >= maxDDPct and not halted and time >= resumeTime
    halted := true
    if alertsOn
        alert(f_msg("HALT_MAX_DRAWDOWN", 0, close, 0, 0, 0), alert.freq_once_per_bar_close)

// ═════════════════════════ POSITION / TRADE STATE ════════════════════════════
bool posOpen = strategy.position_size != 0

// Resolve a finished trade (also catches trades entered and stopped within one bar)
bool resolved = false
if tradeLive and not posOpen and bar_index > signalBar
    tradeLive := false
    resolved  := true
    float pnl = strategy.netprofit - npAtEntry
    if pnl < 0
        consecLoss += 1
        if consecLoss >= maxConsecLoss
            cooldownUntil := bar_index + cooldownBars
            consecLoss := 0
    else
        consecLoss := 0

// ═══════════════════════════════ ENTRY SIGNALS ═══════════════════════════════
bool volOK        = volMA > 0 and volume >= volMA * volMult
bool volatilityOK = atrPct >= atrPctMin and atrPct <= atrPctMax

// A: primary breakout   C: fast breakout (needs stronger ADX and price above the EMA)   B: pullback
bool longBreakout = useBreakout and regimeLong and close > donHi and adxVal >= adxMin and diPlus > diMinus and rsiVal >= rsiMinL and rsiVal <= rsiMaxL and volOK and volatilityOK
bool longFast     = useFast and regimeLong and close > donFastHi and close > ema21 and adxVal >= adxMinFast and diPlus > diMinus and rsiVal >= rsiMinL and rsiVal <= rsiMaxL and volOK and volatilityOK
bool longPullback = usePullback and regimeLong and lowestN <= ema21 + pullTouchAtr * atr14 and close > ema21 and close > high[1] and barsSinceRsiUp <= rsiCrossBars and adxVal >= adxMinPB and volatilityOK

bool shortBreakout = useBreakout and regimeShort and close < donLo and adxVal >= adxMin and diMinus > diPlus and rsiVal >= rsiMinS and rsiVal <= rsiMaxS and volOK and volatilityOK
bool shortFast     = useFast and regimeShort and close < donFastLo and close < ema21 and adxVal >= adxMinFast and diMinus > diPlus and rsiVal >= rsiMinS and rsiVal <= rsiMaxS and volOK and volatilityOK
bool shortPullback = usePullback and regimeShort and highestN >= ema21 - pullTouchAtr * atr14 and close < ema21 and close < low[1] and barsSinceRsiDown <= rsiCrossBars and adxVal >= adxMinPB and volatilityOK

bool goLong  = allowLong  and (longBreakout or longFast or longPullback)
bool goShort = allowShort and (shortBreakout or shortFast or shortPullback)
string entryTag = goLong ? (longBreakout ? "BRK" : longFast ? "FBRK" : "PB") : (shortBreakout ? "BRK" : shortFast ? "FBRK" : "PB")

bool inRange   = not useDateRange or (time >= startTime and time <= endTime)
bool blackout  = (blk1End > blk1Start and time >= blk1Start and time < blk1End) or (blk2End > blk2Start and time >= blk2Start and time < blk2End)
bool canEnter  = barstate.isconfirmed and tradingEnabled and inRange and dataOK and not halted and not dayLocked and bar_index > cooldownUntil and entriesToday < maxEntriesDay and not blackout and not posOpen and not tradeLive and not resolved

int  sigDir    = canEnter ? (goLong ? 1 : goShort ? -1 : 0) : 0
bool signalNow = false

if sigDir != 0
    // --- volatility-based sizing ---
    float riskEff = riskPct * (ddPct >= ddScaleStart ? ddScaleFactor : 1.0)
    float dist    = distNow
    float qtyRisk = strategy.equity * riskEff / 100 / dist
    float qtyCap  = strategy.equity * maxLev / close
    float posQty  = f_lot(math.min(qtyRisk, qtyCap))
    float sq1     = f_lot(posQty * tp1Pct / 100)
    float sq2     = f_lot(posQty * tp2Pct / 100)
    bool  costOK  = costShareNow <= maxCostShare
    bool  sizeOK  = dist > 0 and costOK and posQty * close >= minNotional and sq1 >= lotStep and sq2 >= lotStep and posQty - sq1 - sq2 >= lotStep
    if sizeOK
        float sStop = math.round_to_mintick(close - sigDir * dist)
        float sTp1  = math.round_to_mintick(close + sigDir * tp1R * dist)
        float sTp2  = math.round_to_mintick(close + sigDir * tp2R * dist)
        string act  = sigDir == 1 ? "ENTRY_LONG" : "ENTRY_SHORT"
        strategy.entry(sigDir == 1 ? "L" : "S", sigDir == 1 ? strategy.long : strategy.short, qty = posQty, comment = entryTag, alert_message = alertsOn ? f_msg(act, posQty, close, sStop, sTp1, sTp2) : "")
        curDir       := sigDir
        stopDist     := dist
        initQty      := posQty
        stage        := 0
        curStop      := sStop
        lastStopSent := sStop
        extremeClose := close
        tradeLive    := true
        signalBar    := bar_index
        npAtEntry    := strategy.netprofit
        entriesToday += 1
        signalNow    := true

// ══════════════════════════ EXIT MANAGEMENT (SL / TP) ════════════════════════
if posOpen or signalNow
    int   d       = curDir
    float R       = stopDist
    float entryPx = posOpen ? strategy.position_avg_price : close
    string eid    = d == 1 ? "L" : "S"
    float q1      = f_lot(initQty * tp1Pct / 100)
    float q2      = f_lot(initQty * tp2Pct / 100)

    if posOpen
        extremeClose := d == 1 ? math.max(extremeClose, close) : math.min(extremeClose, close)
        // Stage detection from remaining size (robust to partial fills)
        float absSz = math.abs(strategy.position_size)
        int st = stage
        if q1 > 0 and absSz <= initQty - 0.5 * q1
            st := 1
        if q2 > 0 and absSz <= initQty - q1 - 0.5 * q2
            st := 2
        stage := math.max(stage, st)

    tp1Line := math.round_to_mintick(entryPx + d * tp1R * R)
    tp2Line := math.round_to_mintick(entryPx + d * tp2R * R)

    // Ratcheting stop: initial → break-even+fees after TP1 → chandelier after TP2
    float cand = entryPx - d * R
    if stage >= 1
        float be = entryPx * (1 + d * beFeePct / 100)
        cand := d == 1 ? math.max(cand, be) : math.min(cand, be)
    if stage >= 2 and not na(atrT)
        float ch = extremeClose - d * trailMult * atrT
        cand := d == 1 ? math.max(cand, ch) : math.min(cand, ch)
    curStop := math.round_to_mintick(d == 1 ? math.max(curStop, cand) : math.min(curStop, cand))

    if stage < 1 and q1 > 0
        strategy.exit("TP1", eid, qty = q1, limit = tp1Line, comment = "TP1", alert_message = alertsOn ? f_msg("EXIT_TP1", q1, tp1Line, curStop, tp1Line, tp2Line) : "")
    if stage < 2 and q2 > 0
        strategy.exit("TP2", eid, qty = q2, limit = tp2Line, comment = "TP2", alert_message = alertsOn ? f_msg("EXIT_TP2", q2, tp2Line, curStop, tp1Line, tp2Line) : "")
    // Stop covers the whole remaining position (runner included)
    strategy.exit("SL", eid, stop = curStop, comment = stage >= 1 ? "TRAIL/BE" : "SL", alert_message = alertsOn ? f_msg("EXIT_STOP", math.abs(strategy.position_size), curStop, curStop, tp1Line, tp2Line) : "")

// ─────────── discretionary exits (executed at next bar open) ────────────────
if posOpen
    int   d     = curDir
    float entryPx = strategy.position_avg_price
    int   barsIn = bar_index - strategy.opentrades.entry_bar_index(0)
    float openR  = (close - entryPx) * d / stopDist
    bool regimeFail = dataOK and ((d == 1 and dClose < dEmaS) or (d == -1 and dClose > dEmaS))
    bool macroFail  = dataOK and useMacro and ((d == 1 and mClose < mEma) or (d == -1 and mClose > mEma))
    bool donFail    = stage >= 2 and not na(donExitLo) and ((d == 1 and close < donExitLo) or (d == -1 and close > donExitHi))
    bool timeStop   = stage == 0 and barsIn >= timeStopBars and openR < timeStopR
    string reason = ""
    if halted
        reason := "MAX_DD_HALT"
    else if not tradingEnabled
        reason := "KILL_SWITCH"
    else if regimeFail or macroFail
        reason := "TREND_FAIL"
    else if donFail
        reason := "DONCHIAN_EXIT"
    else if timeStop
        reason := "TIME_STOP"
    if reason != ""
        strategy.close_all(comment = reason, alert_message = alertsOn ? f_msg("EXIT_" + reason, math.abs(strategy.position_size), close, curStop, tp1Line, tp2Line) : "")

// Stop-move notification for the exchange-side bridge
if alertsOn and posOpen and not na(curStop) and curStop != lastStopSent
    alert(f_msg("UPDATE_STOP", math.abs(strategy.position_size), close, curStop, tp1Line, tp2Line), alert.freq_once_per_bar_close)
    lastStopSent := curStop

// ═══════════════════════════════ PLOTS ═══════════════════════════════════════
plot(dEmaF, "Regime EMA fast (4H)", color.new(color.orange, 0), 1)
plot(dEmaS, "Regime EMA slow (4H)", color.new(color.red, 0), 2)
